set(TRAPS_SANITIZER "none" CACHE STRING "Sanitizer: none, address, undefined, or thread")
set_property(CACHE TRAPS_SANITIZER PROPERTY STRINGS none address undefined thread)

  message(FATAL_ERROR "In-source builds are disabled. Use: cmake -S . -B build")
endif()

function(configure_trap target group)
  set_target_properties(${target} PROPERTIES FOLDER ${group})
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive- /EHsc)
    if(TRAPS_WARNINGS_AS_ERRORS)
      target_compile_options(${target} PRIVATE /WX)
    endif()
    if(TRAPS_SANITIZER STREQUAL "address")
      target_compile_options(${target} PRIVATE /fsanitize=address)
    elseif(NOT TRAPS_SANITIZER STREQUAL "none")
      message(FATAL_ERROR "MSVC supports only TRAPS_SANITIZER=address in this project")
    endif()
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    if(TRAPS_WARNINGS_AS_ERRORS)
      target_compile_options(${target} PRIVATE -Werror)
    endif()
    if(NOT TRAPS_SANITIZER STREQUAL "none")
      target_compile_options(${target} PRIVATE -fsanitize=${TRAPS_SANITIZER} -fno-omit-frame-pointer -g)
      target_link_options(${target} PRIVATE -fsanitize=${TRAPS_SANITIZER})
    endif()
  endif()
endfunction()

function(add_trap group name)
  file(GLOB trap_sources CONFIGURE_DEPENDS ${group}/${name}/*.cpp)
  add_executable(${name} ${trap_sources})
  configure_trap(${name} ${group})

  # Create an _unsafe sibling only when the source contains a guarded unsafe branch.
  set(has_unsafe_branch FALSE)
  foreach(source IN LISTS trap_sources)
    file(READ "${source}" source_text)
    string(FIND "${source_text}" "RUN_UNSAFE_EXAMPLE" unsafe_marker)
    if(NOT unsafe_marker EQUAL -1)
      set(has_unsafe_branch TRUE)
    endif()
  endforeach()
  if(TRAPS_BUILD_UNSAFE AND has_unsafe_branch)
    add_executable(${name}_unsafe ${trap_sources})
    target_compile_definitions(${name}_unsafe PRIVATE RUN_UNSAFE_EXAMPLE=1)
    configure_trap(${name}_unsafe ${group}/Unsafe)
  endif()
endfunction()
